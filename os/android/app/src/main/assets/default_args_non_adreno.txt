--log_level=warn
# First-launch profile for non-Adreno GPUs (Mali, PowerVR...). The standard
# profile is tuned on a Snapdragon 865; mid-range chips such as the Helio G200
# (2x Cortex-A76 + 6x Cortex-A55, Mali-G57 MC2) have far less CPU and GPU
# headroom, so this one asks for fewer draws and less CPU work per frame.
# Display settings (resolution, upscaler, AA, shadows, motion blur) belong to the
# in-game Display menu: lower the resolution and turn AA off there first.
--gta4_dynamic_draw_distance=true
# Keep at 0.80 or above: lower values broke dynamic shadows (see the migrations
# in LibertyActivity).
--gta4_dynamic_draw_distance_min=0.80
--gta4_shadow_distance_scale=1.0
--gta4_reflection_capture_distance=near
# Fewer cars and pedestrians: less CPU work for the few fast cores and fewer draws.
--gta4_traffic_density_scale=0.3
--gta4_parked_car_density_scale=0.3
--gta4_ped_density_scale=0.3
--gta4_scenario_ped_density_scale=0.3
--gta4_profile_native_detailed_cpu=false
--gta4_profile_native_detailed_gpu=false
--gta4_native_max_queued_frames=3
# While frames miss 30 FPS, refresh the environment reflection every other frame.
--gta4_fps_guard=2
